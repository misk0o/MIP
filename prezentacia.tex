\section{Databázový návrh}

\begin{frame}{Databázová schéma}
\begin{center}
$\text{Schéma} = \{ U, T, P, C, S, I \}$
\end{center}

\begin{columns}
\column{0.5\textwidth}
\textbf{Hlavné tabuľky:}
\begin{itemize}
\item \textbf{U}sers - používatelia
\item \textbf{T}asks - úlohy
\item \textbf{P}rojects - projekty
\end{itemize}

\column{0.5\textwidth}
\textbf{Podporné tabuľky:}
\begin{itemize}
\item \textbf{C}ategories - kategórie
\item \textbf{S}essions - Pomodoro
\item \textbf{I}ntegrations - prepojenia
\end{itemize}
\end{columns}

\vspace{0.5cm}
\begin{block}{Kľúčové vlastnosti}
UUID identifikátory, JSONB pre AI dáta, Full-text search
\end{block}
\end{frame}

\begin{frame}{Tabuľka Tasks - Detail}
\begin{columns}
\column{0.5\textwidth}
\textbf{Základné polia:}
\begin{itemize}
\item id (UUID)
\item title, description
\item deadline
\item estimated\_duration
\item status (enum)
\end{itemize}

\column{0.5\textwidth}
\textbf{Špeciálne polia:}
\begin{itemize}
\item priority (1-4)
\item urgency (1-4)
\item ai\_suggestions (JSONB)
\item google\_task\_id
\item google\_calendar\_id
\end{itemize}
\end{columns}

\vspace{0.3cm}
\textbf{Eisenhowerova matica:} priority + urgency $\rightarrow$ kvadrant
\end{frame}

\begin{frame}{Optimalizácia databázy}
\textbf{Indexy pre výkon:}
\begin{itemize}
\item \texttt{idx\_tasks\_user\_deadline} - vyhľadávanie podľa používateľa a deadlinu
\item \texttt{idx\_tasks\_status} - filtrovanie podľa stavu
\item \texttt{idx\_tasks\_fulltext (GIN)} - full-text search v slovenčine
\end{itemize}

\vspace{0.5cm}
\begin{block}{Časová zložitosť}
$O(\log n)$ pre indexované queries, kde $n$ = počet úloh
\end{block}

\vspace{0.3cm}
\textbf{JSONB výhody:}
\begin{itemize}
\item Flexibilné ukladanie AI odporúčaní
\item Bez potreby schema migrations
\item Rýchle queries
\end{itemize}
\end{frame}

\section{Integrácie a AI}

\begin{frame}{Google služby integrácia}
\begin{itemize}
\item \textbf{Google Calendar API}
  \begin{itemize}
  \item Obojsmerná synchronizácia udalostí
  \item Automatické vytvorenie time-blocks
  \item Notifikácie a pripomienky
  \end{itemize}
\item \textbf{Google Tasks API}
  \begin{itemize}
  \item Synchronizácia úloh
  \item Zachovanie existujúcich zoznamov
  \end{itemize}
\item \textbf{OAuth 2.0 autentifikácia}
  \begin{itemize}
  \item Bezpečné prihlásenie
  \item Granulárne permissions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Google Gemini AI}
\textbf{Využitie AI pre:}
\begin{enumerate}
\item \textbf{Automatickú prioritizáciu}
  \begin{itemize}
  \item Analýza deadline, náročnosti a kontextu
  \item Odporúčanie optimálneho času začatia
  \end{itemize}
\item \textbf{Rozdelenie úloh}
  \begin{itemize}
  \item Rozčlenenie veľkých projektov
  \item Odhad času na podúlohy
  \end{itemize}
\item \textbf{Kategorizáciu}
  \begin{itemize}
  \item Automatické zaradenie do projektov
  \item Návrh tagov a štítkov
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Synchronizačná stratégia}
\textbf{Konfliktové riešenie:}
\begin{itemize}
\item Last-Write-Wins (LWW) s timestampmi
\item Detekcia simultánnych zmien
\item Používateľské notifikácie pri konfliktoch
\end{itemize}

\vspace{0.5cm}
\textbf{Optimalizácia API volaní:}
\begin{itemize}
\item Request batching (združovanie požiadaviek)
\item Cache layer pre časté queries
\item Rate limiting handling (1000 req/100s pre Calendar)
\end{itemize}
\end{frame}
